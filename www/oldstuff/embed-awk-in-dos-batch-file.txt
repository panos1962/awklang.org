> This seems to work: -
>
> @echo off
> dir c:\djgpp\bin\awk*
> awk "/^goto pastawk/{x=1;next};/^:pastawk/{x=0};x" %0 | awk -f - %1
> goto pastawk
> #awk script starts here ---------------------------
> BEGIN{
>    print "Awk script running"
>    print "ARGC=",ARGC
>    for (i=0; i<ARGC; i++) print "ARGV["i"]=",ARGV[i]
>    print "Hello World!"
> }
> {print FILENAME,NR,$0}
> END {print "end of awk script"}
> #awk script ends here -----------------------------
> :pastawk
> echo back in MSDOS
>
