I figured it had been done several times by several people. A
few things that I added to my script that you may still find
useful, include actually printing the raw code or comments
rather than returning a count.  This is helpful for scripts
beyond those that are interested in simply the count.  I also
take care to identify "comment-markers" that might be quoted,
which can quickly get hairy once you throw single quotes and
double quotes in the mix.  Consider these strings for example:

    " this is quoted /* comment marker not a real one "
    " this is quoted \" and backquoted /* comment marker not a real one "
    ' this is "quoted"  /* comment marker not a real one '

#!/usr/bin/awk -f
# -- must use gawk to work

# A program to count the number of lines in a C source code file
# Written by David Reiss. Copyright (C) 2001 David Reiss.
# This file may be distrubuted under the conditions of the
# GNU GPL, version 2 or later.

# I know there is another "clines.awk" (or similar) program out there.
# I wish I had had it to help me write this, but I didn't.
# This is all my own work.

# Description
# Lines containing only whitespace are considered "blank lines."
# Lines containing only comments and whitespace are "comment lines."
# Lines beginning with a '#' (with optional leading whitespace) are
#   assumed to be preprocessor (cpp) directives.
# Lines containing only whitespace, (optional) comments, and a single
#   brace (either '{' or '}') are considered "brace lines."  These
#   don't count as "real" lines because there number depends on your
#   brace style.
# Lines containing anything else are "real" lines.

# This program has one major bug: it doesn't recognize string or
# character literals.  If you have a string or character literal
# containing a comment delimiter ("/*", "*/", or "//"), clines.awk
# will interpret this as a real comment delimiter.

BEGIN {
    use_cpp_comments = 1
    # Define some regexps for shorthand
    cb = "/\\*"              # comment begin
    ce = "\\*/"              # comment end
    ws = "[ \t]*"            # whitespace
    ic = "([^*]|\\*+[^/])*"  # all characters in a comment
    fc = cb ic ce            # a full comment
    if(use_cpp_comments)
        fc = "(" fc "|" "//.*$)"

    # Initialize (to set type) and document variables
    incom = 0        # Are we in a multi-line comment as the line starts?
    blanklines = 0   # Number of blank lines seen
    comlines = 0     # Number of lines containing comments and whitespace
                     # only
    cpplines = 0     # Lines containing preprocessor directives
    bracelines = 0   # Lines containing only a single brace
    reallines = 0    # "Real" lines containing meaningful C code
}

$0 ~ "^" ws "$" {    # Even if inside a multi-line comment
    blanklines++
    next
}

# Special case: in a multi-line comment for entire line
incom && $0 !~ ce {
    comlines++
    next
}

incom {    # Starts with multi-line comment, but ends
    sub(ic ce ws, "")    # Get rid of starting comment
    incom = 0    # No longer in a comment
}

{
    # Clear the complete comments and whitespace
    gsub(fc, "")
    gsub(ws, "")
    if( match($0, cb) )    # Does an unterminated comment start?
    {
        incom = 1    # The line ends in the middle of a multi-line comment
        sub(cb ".*$", "")    # Get rid of it before processing
    }
}

/^$/    {    # The line is now blank; it must have had only comments
    comlines++
    next
}

/^#/    {    # The line starts with a '#'; it must be a preprocessor
             # directive
    cpplines++
    next
}

/^(\}|\{)$/    {    # The line contains only a single brace; not a
                    # "real" line
    bracelines++
    next
}

{    # Any other text?  The line must contain real C code (horrors)
    reallines++
}

END {    # Just print a report (all counters were initialized to 0)
    print "Blank lines:\t", blanklines
    print "Comment lines:\t", comlines
    print "Cpp lines:\t", cpplines
    print "Brace lines:\t", bracelines
    print "\"Real\" lines:\t", reallines
    print "Total lines:\t",
             blanklines + comlines + cpplines + bracelines + reallines
}
